88 const std::function<bool(const Result<lib::spt::page<T>> &)> &callback)
89 {
90 const auto api_url = lib::strings::starts_with(url, "https://")
91 ? lib::spt::to_relative_url(url)
92 : url;
93
94 get<nlohmann::json>(api_url, [this, key, callback](const Result<nlohmann::json> &result)
95 {
96 if (!result.success())
97 {
98 const QString message = parseErrorMessage(result.message());
99 callback(Result<lib::spt::page<T>>::fail(message));
100 return;
101 }
102
103 const auto &json = result.value();
104 if (!key.empty() && !json.contains(key))
105 {
106 const QString message = QStringLiteral("No such key: %1")
107 .arg(QString::fromStdString(key));
108
109 callback(Result<lib::spt::page<T>>::fail(message));
110 return;
111 }
112
113 lib::spt::page<T> page;
114 try
115 {
116 page = key.empty() ? json : json.at(key);
117 }
118 catch (const std::exception &exception)
119 {
120 const QString message = QString::fromStdString(exception.what());
121 callback(Result<lib::spt::page<T>>::fail(message));
122 return;
123 }
124
125 if (!callback(Result<lib::spt::page<T>>::ok(page))
126 || !page.has_next())
127 {
128 return;
129 }
130
131 get_page(page.next, key, callback);
132 });
133 }

260 template<typename T>
261 static auto parseJson(const QByteArray &data) -> Result<T>
262 {
263 if (data.isEmpty())
264 {
265 return Result<T>::fail(QStringLiteral("No data"));
266 }
267
268 QJsonParseError parseError;
269 const QJsonDocument json = QJsonDocument::fromJson(data, &parseError);
270 if (json.isNull())
271 {
272 return Result<T>::fail(parseError.errorString());
273 }
274
275 if (SpotifyErrorUtil::isErrorObject(json.object()))
276 {
277 return Result<T>::fail(SpotifyErrorUtil::errorMessage(json.object()));
278 }
279
280 return Result<T>::ok(T::fromJson(json));
281 }
