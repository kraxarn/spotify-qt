210 template<typename T>
211 static auto parseJson(const QByteArray &data) -> Result<T>
212 {
213 if (data.isEmpty())
214 {
215 return Result<T>::fail(QStringLiteral("No data"));
216 }
217
218 QJsonParseError parseError;
219 const QJsonDocument json = QJsonDocument::fromJson(data, &parseError);
220 if (json.isNull())
221 {
222 return Result<T>::fail(parseError.errorString());
223 }
224
225 if (SpotifyErrorUtil::isErrorObject(json.object()))
226 {
227 return Result<T>::fail(SpotifyErrorUtil::errorMessage(json.object()));
228 }
229
230 return Result<T>::ok(T::fromJson(json));
231 }
